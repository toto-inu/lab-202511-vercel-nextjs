# プラン機能

## 概要

Free/Pro/Enterpriseの3段階プラン設計。各プランで利用可能な機能と制限を定義。

## プラン一覧

**価格**: 無料

- プロジェクト内Todo: **100個まで**
- ストレージ: 100MB

**機能**:
- ✅ 基本的なTodo管理

## 制限チェック実装

### ユーティリティ関数

```typescript
// lib/plan-utils.ts
import { prisma } from '@/lib/prisma';
import { Plan } from '@prisma/client';

/**
 * Tenantの現在のプランを取得
 */
export async function getTenantPlan(tenantId: string): Promise<Plan> {
  const tenant = await prisma.tenant.findUnique({
    where: { id: tenantId },
    include: { plan: true }
  });

  if (!tenant) {
    throw new Error('Tenant not found');
  }

  return tenant.plan;
}

/**
 * 機能が有効かチェック
 */
export function hasFeature(plan: Plan, featureName: string): boolean {
  const features = plan.features as Record<string, boolean>;
  return features[featureName] === true;
}

/**
 * リソース制限をチェック
 */
export async function checkResourceLimit(
  tenantId: string,
  resourceType: 'projects' | 'users' | 'todos',
  projectId?: string
): Promise<{ allowed: boolean; current: number; limit: number | null }> {
  const plan = await getTenantPlan(tenantId);

  let current: number;
  let limit: number | null;

  switch (resourceType) {
    case 'projects':
      current = await prisma.project.count({ where: { tenantId } });
      limit = plan.maxProjects;
      break;

    case 'users':
      current = await prisma.tenantMember.count({ where: { tenantId } });
      limit = plan.maxUsersPerTenant;
      break;

    case 'todos':
      if (!projectId) throw new Error('projectId required for todos limit');
      current = await prisma.todo.count({
        where: { tenantId, projectId }
      });
      limit = plan.maxTodosPerProject;
      break;

    default:
      throw new Error(`Unknown resource type: ${resourceType}`);
  }

  const allowed = limit === null || current < limit;

  return { allowed, current, limit };
}

/**
 * 制限到達時のエラーメッセージ
 */
export function getLimitErrorMessage(
  resourceType: string,
  limit: number | null,
  planName: string
): string {
  if (limit === null) {
    return '予期しないエラーが発生しました';
  }

  const messages = {
    projects: `プロジェクト数の上限（${limit}個）に達しました。`,
    users: `ユーザー数の上限（${limit}人）に達しました。`,
    todos: `このプロジェクトのTodo数の上限（${limit}個）に達しました。`
  };

  const baseMessage = messages[resourceType as keyof typeof messages];

  if (planName === 'FREE') {
    return `${baseMessage} Proプランにアップグレードして制限を拡張できます。`;
  }

  if (planName === 'PRO') {
    return `${baseMessage} Enterpriseプランにアップグレードして無制限にできます。`;
  }

  return baseMessage;
}
```

### Server Actionでの使用例

```typescript
// actions/project.ts
import { checkResourceLimit, getLimitErrorMessage, getTenantPlan } from '@/lib/plan-utils';

export async function createProject(tenantId: string, data: ProjectData) {
  // 権限チェック
  await requireTenantRole(tenantId, ['OWNER', 'ADMIN']);

  // プラン制限チェック
  const limitCheck = await checkResourceLimit(tenantId, 'projects');

  if (!limitCheck.allowed) {
    const plan = await getTenantPlan(tenantId);
    const message = getLimitErrorMessage('projects', limitCheck.limit, plan.name);
    throw new Error(message);
  }

  // プロジェクト作成
  const project = await prisma.project.create({
    data: {
      ...data,
      tenantId
    }
  });

  return project;
}
```

// components/ResourceLimitBadge.tsx
export async function ResourceLimitBadge({ tenantId, type }: Props) {
  const { current, limit } = await checkResourceLimit(tenantId, type);

  const percentage = limit ? (current / limit) * 100 : 0;
  const isNearLimit = percentage > 80;

  return (
    <div className={isNearLimit ? 'text-orange-600' : 'text-gray-600'}>
      {current} / {limit ?? '無制限'}
    </div>
  );
}

function isTrialExpired(tenant: Tenant): boolean {
