# 認証認可フロー

## 概要

Clerkを利用したマルチテナント対応の認証認可システム設計。

### ロール管理方針

**重要**: ロール情報はClerk metadataではなく、**データベース**で管理します。

- **Clerk**: 認証のみを担当（ユーザーID、メールアドレス、プロフィール情報）
- **Database (Prisma)**: ロール・権限の管理を担当
  - TenantMember: Tenant Role (OWNER/ADMIN/MEMBER)
  - ProjectMember: Project Role (ADMIN/EDITOR/VIEWER)
  - User: グローバル管理者フラグ (isGlobalAdmin)

この設計により、権限変更がリアルタイムで反映され、Clerk側の再認証が不要になります。

## アーキテクチャ

### 全体フロー

```
User (Browser)
    ↓
Clerk SDK (認証)
    ↓
Next.js Middleware (ルート保護)
    ↓
Server Component / Server Action
    ↓
Auth Utils (認証情報取得)
    ↓
Permission Check (権限確認)
    ↓
Prisma (データ操作)
```

### ユーザー同期（拡張）

Clerk認証後、DBユーザー＋デフォルトTenantメンバーシップを作成。

```typescript
// lib/auth-utils.ts
import { currentUser } from '@clerk/nextjs/server';
import { prisma } from '@/lib/prisma';

/**
 * 現在のユーザーを取得（Clerk + DB）
 */
export async function getCurrentUser() {
  const clerkUser = await currentUser();

  if (!clerkUser) {
    throw new Error('Not authenticated');
  }

  // DB上のユーザーを検索
  let user = await prisma.user.findUnique({
    where: { clerkId: clerkUser.id }
  });

  // 初回ログイン時：ユーザー作成
  if (!user) {
    user = await prisma.user.create({
      data: {
        clerkId: clerkUser.id,
        email: clerkUser.emailAddresses[0].emailAddress,
        name: clerkUser.fullName || clerkUser.firstName,
        avatarUrl: clerkUser.imageUrl
      }
    });

    // デフォルトTenant作成（個人用）
    await createDefaultTenant(user.id);
  }

  return user;
}

/**
 * 初回ログイン時にデフォルトTenantを作成
 */
async function createDefaultTenant(userId: string) {
  const user = await prisma.user.findUnique({ where: { id: userId } });
  if (!user) throw new Error('User not found');

  // FreeプランのIDを取得
  const freePlan = await prisma.plan.findUnique({
    where: { name: 'FREE' }
  });

  if (!freePlan) throw new Error('Free plan not found');

  // 個人用Tenant作成
  const tenant = await prisma.tenant.create({
    data: {
      name: `${user.name}'s Workspace`,
      slug: `${user.email.split('@')[0]}-${Date.now()}`,
      planId: freePlan.id,
      status: 'ACTIVE'
    }
  });

  // ユーザーをOWNERとして追加
  await prisma.tenantMember.create({
    data: {
      userId: user.id,
      tenantId: tenant.id,
      role: 'OWNER',
      joinedAt: new Date()
    }
  });

  return tenant;
}
```

## 2. 認可フロー（マルチテナント対応）

### 現在のTenantコンテキスト取得

ユーザーがどのTenantで作業しているかを管理。

```typescript
// lib/auth/tenant-context.ts
import { cookies } from 'next/headers';

/**
 * 現在アクティブなTenantIDを取得
 */
export async function getCurrentTenantId(): Promise<string | null> {
  const cookieStore = await cookies();
  return cookieStore.get('current-tenant-id')?.value || null;
}

/**
 * アクティブTenantを設定
 */
export async function setCurrentTenantId(tenantId: string) {
  const cookieStore = await cookies();
  cookieStore.set('current-tenant-id', tenantId, {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    maxAge: 60 * 60 * 24 * 30 // 30日
  });
}

/**
 * 現在のTenantを取得（メンバーシップチェック付き）
 */
export async function getCurrentTenant() {
  const user = await getCurrentUser();
  const tenantId = await getCurrentTenantId();

  if (!tenantId) {
    // デフォルトTenantを取得
    const membership = await prisma.tenantMember.findFirst({
      where: { userId: user.id },
      include: { tenant: { include: { plan: true } } },
      orderBy: { joinedAt: 'asc' }
    });

    if (!membership) {
      throw new Error('No tenant found for user');
    }

    // Cookieに保存
    await setCurrentTenantId(membership.tenantId);

    return membership.tenant;
  }

  // 指定されたTenantへのアクセス権をチェック
  const membership = await prisma.tenantMember.findUnique({
    where: {
      userId_tenantId: { userId: user.id, tenantId }
    },
    include: { tenant: { include: { plan: true } } }
  });

  if (!membership) {
    throw new Error('Access denied to tenant');
  }

  return membership.tenant;
}

/**
 * ユーザーが所属する全Tenantを取得
 */
export async function getUserTenants() {
  const user = await getCurrentUser();

  const memberships = await prisma.tenantMember.findMany({
    where: { userId: user.id },
    include: {
      tenant: {
        include: { plan: true }
      }
    },
    orderBy: { joinedAt: 'asc' }
  });

  return memberships.map(m => ({
    ...m.tenant,
    role: m.role
  }));
}
```

### 権限チェックヘルパー

```typescript
// lib/auth/permission.ts
import { TenantRole, ProjectRole } from '@prisma/client';

/**
 * Tenantへのアクセスを要求
 */
export async function requireAuth() {
  const user = await getCurrentUser();
  return user;
}

/**
 * 特定TenantRoleを要求
 */
export async function requireTenantRole(
  tenantId: string,
  allowedRoles: TenantRole[]
) {
  const user = await getCurrentUser();

  const membership = await prisma.tenantMember.findUnique({
    where: {
      userId_tenantId: { userId: user.id, tenantId }
    }
  });

  if (!membership || !allowedRoles.includes(membership.role)) {
    throw new Error('Insufficient tenant permissions');
  }

  return { user, role: membership.role };
}

/**
 * ProjectRoleを要求
 */
export async function requireProjectRole(
  projectId: string,
  allowedRoles: ProjectRole[]
) {
  const user = await getCurrentUser();

  const project = await prisma.project.findUnique({
    where: { id: projectId }
  });

  if (!project) {
    throw new Error('Project not found');
  }

  // Tenant OWNER/ADMINは自動的にプロジェクトADMIN
  const tenantMembership = await prisma.tenantMember.findUnique({
    where: {
      userId_tenantId: { userId: user.id, tenantId: project.tenantId }
    }
  });

  if (tenantMembership?.role === 'OWNER' || tenantMembership?.role === 'ADMIN') {
    return { user, role: 'ADMIN' as ProjectRole, project };
  }

  // 明示的なProjectMembershipをチェック
  const projectMembership = await prisma.projectMember.findUnique({
    where: {
      userId_projectId: { userId: user.id, projectId }
    }
  });

  if (!projectMembership || !allowedRoles.includes(projectMembership.role)) {
    throw new Error('Insufficient project permissions');
  }

  return { user, role: projectMembership.role, project };
}

/**
 * Todoの編集・削除権限チェック
 */
export async function canEditTodo(userId: string, todoId: string): Promise<boolean> {
  const todo = await prisma.todo.findUnique({
    where: { id: todoId },
    include: { project: true }
  });

  if (!todo) return false;

  // 作成者は常に編集可能
  if (todo.createdById === userId) return true;

  // Project権限をチェック
  try {
    await requireProjectRole(todo.projectId, ['ADMIN']);
    return true;
  } catch {
    return false;
  }
}

/**
 * グローバル管理者チェック
 */
export async function requireGlobalAdmin() {
  const user = await getCurrentUser();

  if (!user.isGlobalAdmin) {
    throw new Error('Global admin access required');
  }

  return user;
}
```

## 3. Server Actionでの実装例

### Todo作成

```typescript
// actions/todo.ts
'use server'

import { requireProjectRole } from '@/lib/auth/permission';
import { checkResourceLimit } from '@/lib/plan-utils';

export async function createTodo(projectId: string, data: TodoData) {
  // 1. 認証＋権限チェック
  const { user, project } = await requireProjectRole(projectId, ['ADMIN', 'EDITOR']);

  // 2. プラン制限チェック
  const limitCheck = await checkResourceLimit(
    project.tenantId,
    'todos',
    projectId
  );

  if (!limitCheck.allowed) {
    throw new Error(
      `Todoの上限（${limitCheck.limit}個）に達しました。プランをアップグレードしてください。`
    );
  }

  // 3. Todo作成
  const todo = await prisma.todo.create({
    data: {
      ...data,
      tenantId: project.tenantId,
      projectId: projectId,
      createdById: user.id
    }
  });

  revalidatePath(`/projects/${projectId}`);

  return todo;
}
```

### Todo編集

```typescript
export async function updateTodo(todoId: string, data: Partial<TodoData>) {
  const user = await requireAuth();

  // 編集権限チェック
  const canEdit = await canEditTodo(user.id, todoId);

  if (!canEdit) {
    throw new Error('このTodoを編集する権限がありません');
  }

  const todo = await prisma.todo.update({
    where: { id: todoId },
    data
  });

  revalidatePath(`/projects/${todo.projectId}`);

  return todo;
}
```

### プロジェクト作成

```typescript
// actions/project.ts
export async function createProject(tenantId: string, data: ProjectData) {
  // 1. Tenant権限チェック
  const { user } = await requireTenantRole(tenantId, ['OWNER', 'ADMIN']);

  // 2. プラン制限チェック
  const limitCheck = await checkResourceLimit(tenantId, 'projects');

  if (!limitCheck.allowed) {
    throw new Error(
      `プロジェクトの上限（${limitCheck.limit}個）に達しました。`
    );
  }

  // 3. プロジェクト作成
  const project = await prisma.project.create({
    data: {
      ...data,
      tenantId
    }
  });

  // 4. 作成者をプロジェクトADMINとして追加
  await prisma.projectMember.create({
    data: {
      userId: user.id,
      projectId: project.id,
      role: 'ADMIN'
    }
  });

  revalidatePath('/projects');

  return project;
}
```

## 4. Prisma Middleware（Tenant分離強制）

全クエリでtenantIdフィルタを強制。

**注意**: Prisma 5以降では`$use`の代わりに`$extends`を使用します。

```typescript
// lib/prisma-tenant.ts
import { Prisma } from '@prisma/client';
import { prisma } from './prisma';

/**
 * Tenant分離を強制するPrisma拡張
 */
export function createTenantPrisma(tenantId: string) {
  return prisma.$extends({
    name: 'tenantIsolation',
    query: {
      todo: {
        async $allOperations({ operation, model, args, query }) {
          // tenantIdを自動追加
          if (operation === 'create' || operation === 'createMany') {
            if (Array.isArray(args.data)) {
              args.data = args.data.map(d => ({ ...d, tenantId }));
            } else {
              args.data = { ...args.data, tenantId };
            }
          }

          // whereにtenantIdを自動追加
          if (['findMany', 'findFirst', 'findUnique', 'count', 'update', 'updateMany', 'delete', 'deleteMany'].includes(operation)) {
            args.where = { ...args.where, tenantId };
          }

          return query(args);
        }
      },
      project: {
        async $allOperations({ operation, args, query }) {
          // Projectにも同様の処理
          if (operation === 'create' || operation === 'createMany') {
            if (Array.isArray(args.data)) {
              args.data = args.data.map(d => ({ ...d, tenantId }));
            } else {
              args.data = { ...args.data, tenantId };
            }
          }

          if (['findMany', 'findFirst', 'findUnique', 'count', 'update', 'updateMany', 'delete', 'deleteMany'].includes(operation)) {
            args.where = { ...args.where, tenantId };
          }

          return query(args);
        }
      }
    }
  });
}

// 使用例（Server Action内）
export async function getTodos(projectId: string) {
  const tenant = await getCurrentTenant();
  const db = createTenantPrisma(tenant.id);

  return await db.todo.findMany({
    where: { projectId } // tenantIdは自動で追加される
  });
}
```

**代替案**: すべてのクエリで明示的に`tenantId`を指定する方法もあります。こちらの方がシンプルで、予期しない動作を防げます。

```typescript
// より安全な明示的アプローチ（推奨）
export async function getTodos(projectId: string) {
  const tenant = await getCurrentTenant();

  return await prisma.todo.findMany({
    where: {
      projectId,
      tenantId: tenant.id // 明示的にtenantIdを指定
    }
  });
}
```

## 6. セキュリティ考慮事項

### 1. Tenant漏洩防止

- **すべてのクエリでtenantIdフィルタ必須**
- Prisma Middlewareで自動付与（または明示的に指定）
- Server Actionで明示的にチェック
- テスト環境で異なるTenant間のデータ漏洩を検証

### 2. CSRF対策

- Next.jsのServer Action（POST）で自動対応
- Clerk SDKのCSRF保護
- SameSite Cookie属性の設定

### 3. セッション管理

- ClerkのJWTトークン（短期有効期限推奨）
- HTTPOnly Cookie
- Secure Cookie（本番環境）
- セッションタイムアウト: 30分（アイドル時）

### 5. APIアクセス（将来）

```typescript
// model ApiToken
model ApiToken {
  id        String   @id @default(cuid())
  token     String   @unique
  name      String   // "Production API", "Staging API"
  tenantId  String
  tenant    Tenant   @relation(fields: [tenantId], references: [id])

  scopes    String[] // ["read:todos", "write:todos", "read:projects"]
  expiresAt DateTime?
  lastUsedAt DateTime?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

// APIトークン認証
export async function authenticateApiToken(token: string) {
  const apiToken = await prisma.apiToken.findUnique({
    where: { token },
    include: { tenant: true }
  });

  if (!apiToken) {
    throw new Error('Invalid API token');
  }

  if (apiToken.expiresAt && apiToken.expiresAt < new Date()) {
    throw new Error('API token expired');
  }

  // 最終使用日時を更新
  await prisma.apiToken.update({
    where: { id: apiToken.id },
    data: { lastUsedAt: new Date() }
  });

  return apiToken;
}

// API Routeでの使用例
export async function GET(req: NextRequest) {
  const authHeader = req.headers.get('Authorization');
  if (!authHeader?.startsWith('Bearer ')) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const token = authHeader.substring(7);
  const apiToken = await authenticateApiToken(token);

  // スコープチェック
  if (!apiToken.scopes.includes('read:todos')) {
    return NextResponse.json({ error: 'Insufficient scope' }, { status: 403 });
  }

  // Tenant contextで処理
  const todos = await prisma.todo.findMany({
    where: { tenantId: apiToken.tenantId }
  });

  return NextResponse.json(todos);
}
```

### 6. CORS設定（APIアクセス時）

```typescript
// middleware.ts
import { NextResponse } from 'next/server';

export function middleware(req: NextRequest) {
  // API routesにのみCORSヘッダーを追加
  if (req.nextUrl.pathname.startsWith('/api/')) {
    const res = NextResponse.next();

    res.headers.set('Access-Control-Allow-Origin', process.env.ALLOWED_ORIGINS || '*');
    res.headers.set('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS');
    res.headers.set('Access-Control-Allow-Headers', 'Content-Type, Authorization');
    res.headers.set('Access-Control-Max-Age', '86400');

    return res;
  }

  return NextResponse.next();
}
```

### 8. 監査ログ（Enterprise機能）

重要な操作を記録:

```typescript
model AuditLog {
  id        String   @id @default(cuid())
  tenantId  String
  tenant    Tenant   @relation(fields: [tenantId], references: [id])

  userId    String
  user      User     @relation(fields: [userId], references: [id])

  action    String   // "todo.create", "project.delete", "user.invite"
  resource  String   // "Todo:123", "Project:456"
  metadata  Json     // { before: {...}, after: {...} }

  ipAddress String?
  userAgent String?

  createdAt DateTime @default(now())

  @@index([tenantId, createdAt])
  @@index([userId])
}

// 監査ログ記録
export async function logAudit(params: {
  tenantId: string;
  userId: string;
  action: string;
  resource: string;
  metadata?: any;
  req?: NextRequest;
}) {
  // Enterprise Planのみ記録
  const tenant = await prisma.tenant.findUnique({
    where: { id: params.tenantId },
    include: { plan: true }
  });

  if (!hasFeature(tenant.plan, 'auditLog')) {
    return; // 監査ログ機能が無効
  }

  await prisma.auditLog.create({
    data: {
      ...params,
      ipAddress: params.req?.headers.get('x-forwarded-for') || params.req?.ip,
      userAgent: params.req?.headers.get('user-agent')
    }
  });
}
```
